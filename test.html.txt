  <script>
    gsap.registerPlugin(ScrollTrigger);

    // Initialize ScrollTrigger Timeline
    const tl = gsap.timeline({
      scrollTrigger: {
        trigger: "#heroWrapper",
        start: "top top",
        end: "bottom bottom",
        scrub: 1, // Smooth scrub effect (1 second lag)
      }
    });

    // 1. Text fades out quickly and moves upwards
    tl.to("#heroContent", {
      opacity: 0,
      scale: 0.8,
      y: -100,
      duration: 1,
      ease: "power2.inOut"
    }, 0)

    // 2. Window Frame zooms way in (creates the effect of passing through the window)
    .to("#windowFrame", {
      scale: 5, // Zooms window bounds past edge of screen
      transformOrigin: "center center",
      opacity: 0, // Soft fade out at the tail end
      duration: 3,
      ease: "power2.inOut"
    }, 0)

    // 3. Parallax effect on Sky Background (Scales down and shifts up)
    .to("#skyBg", {
      scale: 1.0, // Scale down to natural full-width ratio
      y: -50,
      filter: "brightness(1.1) contrast(1.0)",
      duration: 3,
      ease: "power1.out"
    }, 0);
